<!-- ==========================================================================
     THE SERVER (THE PAIR ROAD). src/js/pair.js is the client; this element is
     the generator's server code, and IT IS PUBLIC SOURCE - anyone who views this
     generator reads every word of it. Nothing secret may ever be put here, and
     nothing secret ever is: the whole job below is to relay opaque frames
     between two devices that already share a code. The server cannot read them
     (every frame is AES-GCM ciphertext the devices made), cannot derive a room
     (a room is named by the SHA-256 of the shared code, and the code itself
     never crosses the wire), and cannot keep one (rooms live in the platform's
     own pub/sub registry and in nothing else - `state`, the durable 50MiB, IS
     NOT TOUCHED AT ALL, so there is no per-user storage here and nothing to
     migrate when this file changes).
     ========================================================================== -->
<script type="text/x-server-plugin">
// THE WHOLE SERVER IS FAN-OUT AND TWO QUESTIONS.
//
// The terminal's PAIR road (src/js/pair.js) needs no authority, no directory and no durable memory.
// Two devices that share a code derive the SAME room id - the SHA-256 of that code - and both
// subscribe to the topic below; the platform's own pub/sub then hands each device the other's frames
// (the sender sees its own frame come back as well, which is how a device knows the room is live).
// Frames are CIPHERTEXT: each one is sealed by the sending device with a key derived from the code,
// which the server never learns.
//
// What is left for the server is therefore only: check the room id is the shape of a hash, and pass
// the frame on. Nothing else is kept between events except one small rate bucket per connection, and
// that is allowed to be lost - a rebuilt server (the platform may rebuild these globals at any time)
// is FUNCTIONALLY IDENTICAL, because a device re-subscribes on its own every twenty seconds.
//
// The two RPCs exist only because pub/sub cannot answer them: how many devices are in a room, and
// whether the room is alive at all - which is how a device tells "nobody has joined yet" apart from
// "that code is wrong". Nothing here trusts the client: a room id that is not a 64-character
// lowercase hex string is refused, a frame that is empty or over the cap is refused, and a device
// must be SUBSCRIBED to the room it is publishing into (the platform's own registry is the check -
// `conn.isSubscribed`), so this cannot be used as an open relay for arbitrary topics.
var PAIR_PREFIX = "liapair-";
var PAIR_MAX_FRAME = 24000;        // characters of ciphertext; the client chunks at about 12KB
var PAIR_RATE_PER_SEC = 60;        // frames a second a connection may relay
var PAIR_RATE_BURST = 120;
var US = "\u001f";                 // the field separator the client also uses, so neither side needs JSON

// ON THERE BEING NO ROOM CEILING HERE: the platform already bounds this - 4,096 topics per generator,
// 64 subscriptions per connection, 10,000 connections - and a topic cannot outlive its subscribers, so
// a room that is closed stops existing on its own. Counting rooms in this script would be bookkeeping
// that the runtime rebuilds away in any case (see the header), so the bound that is actually enforced
// is the per-connection frame rate, the frame size, and the platform's own limits.

function pairTopic(roomId) {
  var s = String(roomId == null ? "" : roomId).toLowerCase();
  if (s.length !== 64) return "";
  for (var i = 0; i < 64; i++) {
    var c = s.charCodeAt(i);
    if (!((c >= 48 && c <= 57) || (c >= 97 && c <= 102))) return "";   // 0-9 and a-f only
  }
  return PAIR_PREFIX + s;
}
function pairSubscribers(topic) { try { return pubsub.numSubscribers(topic); } catch (e) { return 0; } }

// A TOKEN BUCKET PER CONNECTION, and nothing else. It is deliberately small and deliberately
// forgettable: its only job is to stop one client turning this relay into a flood.
var pairBuckets = new Map();
function pairAllow(conn, now) {
  var b = pairBuckets.get(conn.id);
  if (!b) { b = { c: PAIR_RATE_BURST, t: now }; pairBuckets.set(conn.id, b); }
  else { b.c = Math.min(PAIR_RATE_BURST, b.c + ((now - b.t) / 1000) * PAIR_RATE_PER_SEC); b.t = now; }
  if (b.c < 1) return false;
  b.c -= 1;
  return true;
}

function pairJoin(conn, roomId) {
  var t = pairTopic(roomId);
  if (!t) return "err" + US + "badroom";
  if (!pairAllow(conn, Date.now())) return "err" + US + "rate";
  try { conn.subscribe(t); } catch (e) { return "err" + US + "subscribe"; }
  // THE SENDER'S OWN FRAME COMES BACK TOO (the platform includes the sender in a publish), and that
  // is on purpose: it is the ack that says the room is really working.
  return "ok" + US + pairSubscribers(t);
}

function pairLeave(conn, roomId) {
  var t = pairTopic(roomId);
  if (!t) return "err" + US + "badroom";
  try { conn.unsubscribe(t); } catch (e) {}
  return "ok";
}

// ONE FRAME IN, ONE PUBLISH OUT. The payload is roomId + US + frame, so the server needs NO memory
// of which room a connection is in - which is what makes it immune to the platform rebuilding these
// globals under it.
self.onmessage = ({ conn, data }) => {
  try {
    if (typeof data !== "string") return;
    var i = data.indexOf(US);
    if (i < 1) return;
    var roomId = data.slice(0, i), frame = data.slice(i + 1);
    var t = pairTopic(roomId);
    if (!t || !frame || frame.length > PAIR_MAX_FRAME) return;
    if (!conn.isSubscribed(t)) { try { conn.send("err" + US + "notjoined"); } catch (e) {} return; }
    if (!pairAllow(conn, Date.now())) { try { conn.send("err" + US + "rate"); } catch (e) {} return; }
    pubsub.publish(t, data.slice(i + 1));   // the sender is included: it is the device's own ack
  } catch (e) {}
};

self.onclose = ({ conn }) => {
  try { pairBuckets.delete(conn.id); } catch (e) {}
};

self.rpc = {
  // JOIN (or re-join) A ROOM. Idempotent by design: a device calls it again every twenty seconds, so
  // a room that was lost to a rebuild is back before anyone notices.
  pair_join({ conn }, data) {
    try { return pairJoin(conn, String(data)); } catch (e) { return "err" + US + "internal"; }
  },
  pair_leave({ conn }, data) {
    try { return pairLeave(conn, String(data)); } catch (e) { return "err"; }
  },
  // HOW MANY DEVICES ARE IN THE ROOM - the platform's own count of who holds the topic open.
  pair_count({ conn }, data) {
    try { var t = pairTopic(String(data)); if (!t) return "err" + US + "badroom"; return "ok" + US + pairSubscribers(t); }
    catch (e) { return "ok" + US + "0"; }
  },
  // IS ANYTHING THERE? A device that is about to join asks first, so "nobody has joined yet" and
  // "that code is wrong" are different sentences on the screen.
  pair_alive({ conn }, data) {
    try { var t = pairTopic(String(data)); if (!t) return "err" + US + "badroom"; return pairSubscribers(t) > 0 ? "yes" : "no"; }
    catch (e) { return "no"; }
  },
  // A LIVENESS PING, and the only thing this server answers without a room: it is what the client
  // uses to tell a dead socket from a quiet room. (It still takes a payload field - every rpc here
  // takes exactly one argument.)
  pair_ping({ conn }, data) { return "pong"; }
};
</script>

<script>
/* THE STARTUP WATCH (plain ES5 - it has to run even when everything else does not).

   The terminal is a shell of src/ files, and a src/ file is only on the server once the
   generator has been SAVED. While editing, the page serves the unsaved ones locally, so the
   terminal looks whole on the machine that built it and dies silently everywhere else - a
   tablet, a phone, the public URL - with no error worth reading, because a script that 404s
   never gets to complain in a useful way. This watches for exactly that, keeps what it saw
   for `/backup diag`, and if the terminal cannot start, says so on the screen instead of
   leaving an empty page. */
(function () {
  var issues = [];
  window.__bootIssues = issues;

  document.addEventListener("error", function (e) {
    var t = e.target;
    if (!t) return;
    if (t.tagName === "SCRIPT" && t.getAttribute("src")) {
      issues.push("SCRIPT DID NOT LOAD :: " + t.getAttribute("src"));
    } else if (t.tagName === "LINK" && t.getAttribute("href")) {
      issues.push("STYLESHEET DID NOT LOAD :: " + t.getAttribute("href"));
    }
  }, true);

  window.addEventListener("error", function (e) {
    var at = e && e.filename ? " [" + String(e.filename).split("/").slice(-1)[0] + ":" + (e.lineno || 0) + "]" : "";
    issues.push("ERROR :: " + ((e && e.message) || "unknown") + at);
  });
  window.addEventListener("unhandledrejection", function (e) {
    var r = e && e.reason;
    issues.push("UNHANDLED :: " + ((r && (r.message || r)) || "unknown"));
  });

  /* THE SECOND ASK, FOR EVERYTHING ELSE THE TERMINAL FETCHES. Not every src/ file is a
     <script> tag: the command list, the memory index, the trait pack and the rest are
     fetched by the modules themselves, and a load that races the file storage can lose one
     of those as easily as a script - which is how a terminal can come up whole and still
     have /help empty. Only a GET for a src/ path is touched, only when the first answer is
     an error or a 404, and only once, with the same cache-busting second ask the scripts
     get. Everything else - every other fetch, every other method, and the response itself
     when it finally arrives - is passed through exactly as it was. */
  (function () {
    var realFetch = window.fetch;
    if (!realFetch) return;
    function isSrcPath(u) {
      if (typeof u !== "string" || !u) return false;
      return u.indexOf("src/") === 0 || u.indexOf("/src/") !== -1;
    }
    window.fetch = function (input, init) {
      var self = this, args = arguments;
      if (!isSrcPath(input)) return realFetch.apply(self, args);
      if (init && init.method && String(init.method).toUpperCase() !== "GET") return realFetch.apply(self, args);
      var attempt = function (url, first) {
        return realFetch.call(self, url, init).then(function (res) {
          if (first && (!res || !res.ok)) return wait().then(function () { return attempt(bust(url), false); });
          return res;
        }, function (err) {
          if (!first) throw err;
          return wait().then(function () { return attempt(bust(url), false); });

    var src = el && el.getAttribute && el.getAttribute("src");

<!-- THE DOOR'S OWN FALLBACK (plain ES5, at the very foot of the page). Same argument as THE
     STARTUP WATCH at the top, applied to the one control that must never be a dead end: if
     src/js/agegate.js never arrived - a src/ file is only on the server once the generator is
     SAVED, and a 404 leaves no working module behind - the 18+ checkbox would be a door with no
     handle, and the terminal would be unreachable for everyone including him. This is the "give
     the module a moment" fallback from the plan: it OUTLIVES src/js/agegate.js, wires the same two
     buttons to the same key, and does exactly one thing less than the module (no receipts on the
     stage, no /agegate command). It is not used when the module is there - `window.agegate` is the
     test, and it is set at the foot of that file, above this script in document order. -->
<script>
(function () {
  if (window.agegate) return;            // the real door is there; this is only its shadow
  var gate = document.getElementById("ageGate");
  var box = document.getElementById("ageGateBox");
  var enter = document.getElementById("ageGateEnterBtn");
  var leave = document.getElementById("ageGateLeaveBtn");
  if (!gate || !box || !enter) return;
  try { if (sessionStorage.getItem("lia_dev") === "1") { gate.hidden = true; return; } } catch (e) {}
  function paint() { enter.disabled = !box.checked; }
  function key() { return (window.root && window.root.kv && window.root.kv.lia_settings) || null; }
  box.addEventListener("change", paint);
  box.addEventListener("keydown", function (ev) { if (ev.key === "Enter") { ev.preventDefault(); go(); } });
  enter.addEventListener("click", go);
  if (leave) leave.addEventListener("click", function () { location.href = "https://perchance.org/"; });
  function go() {
    if (!box.checked) { paint(); return; }
    var k = key();
    if (k) { try { k.set("agegate", { ok: true, at: Date.now(), zone: (window.logTime && window.logTime.zone) ? window.logTime.zone() : "" }); } catch (e) {} }
    gate.hidden = true;
  }
  paint();
  (async function () {
    var k = key();
    if (!k) return;
    try { var v = await k.get("agegate"); if (v && v.ok === true) gate.hidden = true; } catch (e) {}
  })();
})();

      for (var i = 0; i < v.length; i++) out += (v[i] < 16 ? "0" : "") + v[i].toString(16);
      return out;
    } catch (e) { return null; }
  }

    var rnd = new Uint8Array(16), out = "", i;
    try { if (window.crypto && window.crypto.getRandomValues) window.crypto.getRandomValues(rnd); else throw 0; }
    catch (e) { for (i = 0; i < 16; i++) rnd[i] = Math.floor(Math.random() * 256); }
    for (i = 0; i < 16; i++) out += ABET.charAt(rnd[i] % ABET.length);
    return out.slice(0, 4) + "-" + out.slice(4, 8) + "-" + out.slice(8, 12) + "-" + out.slice(12, 16);
  }
